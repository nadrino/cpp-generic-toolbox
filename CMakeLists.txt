cmake_minimum_required(VERSION 3.14)
project(cpp-generic-toolbox LANGUAGES CXX)

set(GENERIC_TOOLBOX_STANDALONE OFF)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(GENERIC_TOOLBOX_STANDALONE ON)
endif()

option(GENERIC_TOOLBOX_COMPILED "Compile non-template implementations into module libraries" OFF)
option(GENERIC_TOOLBOX_BUILD_SHARED "Build compiled modules as shared libraries" "${BUILD_SHARED_LIBS}")
option(GENERIC_TOOLBOX_WITH_ROOT "Enable the ROOT extension" OFF)
option(GENERIC_TOOLBOX_WITH_JSON "Enable the nlohmann JSON extension" OFF)
option(GENERIC_TOOLBOX_WITH_YAML "Enable the yaml-cpp extension" OFF)
option(GENERIC_TOOLBOX_BUILD_EXAMPLES "Build the toolbox example" ${GENERIC_TOOLBOX_STANDALONE})
option(GENERIC_TOOLBOX_INSTALL_PACKAGE "Install headers and an exported CMake package" ${GENERIC_TOOLBOX_STANDALONE})
option(GENERIC_TOOLBOX_BUILD_TESTS "Build the toolbox tests" ${GENERIC_TOOLBOX_STANDALONE})

# Declare dependencies per module, including when a parent uses link_libraries().
set_property(DIRECTORY PROPERTY LINK_LIBRARIES "")

add_library(GenericToolbox INTERFACE)
add_library(GenericToolbox::GenericToolbox ALIAS GenericToolbox)
if(GENERIC_TOOLBOX_INSTALL_PACKAGE)
  install(TARGETS GenericToolbox EXPORT GenericToolboxTargets)
endif()

function(addToolboxModule module_)
  string(TOUPPER "${module_}" moduleUpper)
  set(compileModule OFF)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/GenericToolbox.${module_}.cpp")
    set(GENERIC_TOOLBOX_COMPILE_${moduleUpper} AUTO CACHE STRING "Compile ${module_}: AUTO, ON or OFF")
    set_property(CACHE GENERIC_TOOLBOX_COMPILE_${moduleUpper} PROPERTY STRINGS AUTO ON OFF)
    if(GENERIC_TOOLBOX_COMPILE_${moduleUpper} STREQUAL "AUTO")
      set(compileModule ${GENERIC_TOOLBOX_COMPILED})
    elseif(GENERIC_TOOLBOX_COMPILE_${moduleUpper})
      set(compileModule ON)
    endif()
  endif()
  if(compileModule)
    if(GENERIC_TOOLBOX_BUILD_SHARED)
      add_library(GenericToolbox${module_} SHARED src/GenericToolbox.${module_}.cpp)
      target_compile_definitions(GenericToolbox${module_}
        PUBLIC CPP_GENERIC_TOOLBOX_${moduleUpper}_SHARED
        PRIVATE CPP_GENERIC_TOOLBOX_${moduleUpper}_EXPORTS)
    else()
      add_library(GenericToolbox${module_} STATIC src/GenericToolbox.${module_}.cpp)
    endif()
    set(scope PUBLIC)
    target_compile_definitions(GenericToolbox${module_} PUBLIC CPP_GENERIC_TOOLBOX_${moduleUpper}_COMPILED)
    target_compile_features(GenericToolbox${module_} PUBLIC cxx_std_17)
    set_target_properties(GenericToolbox${module_} PROPERTIES POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
  else()
    add_library(GenericToolbox${module_} INTERFACE)
    set(scope INTERFACE)
    target_compile_features(GenericToolbox${module_} INTERFACE cxx_std_14)
  endif()
  add_library(GenericToolbox::${module_} ALIAS GenericToolbox${module_})
  set_target_properties(GenericToolbox${module_} PROPERTIES EXPORT_NAME ${module_})
  target_include_directories(GenericToolbox${module_} ${scope}
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>)
  if(MSVC)
    target_compile_options(GenericToolbox${module_} ${scope} /permissive- /Zc:__cplusplus /Zc:preprocessor /utf-8)
    # The toolbox intentionally uses standard C/POSIX-compatible CRT APIs.
    target_compile_definitions(GenericToolbox${module_} ${scope} _CRT_SECURE_NO_WARNINGS)
  endif()
  foreach(dependency IN LISTS ARGN)
    target_link_libraries(GenericToolbox${module_} ${scope} GenericToolbox::${dependency})
  endforeach()
  target_link_libraries(GenericToolbox INTERFACE GenericToolbox::${module_})
  if(GENERIC_TOOLBOX_INSTALL_PACKAGE)
    install(TARGETS GenericToolbox${module_} EXPORT GenericToolboxTargets
      ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  elseif(compileModule)
    install(TARGETS GenericToolbox${module_}
      ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
  endif()
endfunction()

function(linkToolboxExternalLibraries module_)
  get_target_property(moduleType GenericToolbox${module_} TYPE)
  if(moduleType STREQUAL "INTERFACE_LIBRARY")
    target_link_libraries(GenericToolbox${module_} INTERFACE ${ARGN})
  else()
    target_link_libraries(GenericToolbox${module_} PUBLIC ${ARGN})
  endif()
endfunction()

addToolboxModule(Log Macro)

addToolboxModule(Os String)

  linkToolboxExternalLibraries(Os Psapi)
endif()
addToolboxModule(Fs String)

addToolboxModule(Wrappers Macro)
addToolboxModule(Utils Wrappers Vector String Os)
addToolboxModule(Time Utils)
find_package(Threads REQUIRED)
addToolboxModule(Thread Wrappers Vector Time Macro)
linkToolboxExternalLibraries(Thread Threads::Threads)
addToolboxModule(Stream)
addToolboxModule(Map String)
addToolboxModule(Loops Macro)
addToolboxModule(ZipIterator Macro)
addToolboxModule(Misc)
addToolboxModule(MakeEnum String Vector)

if(GENERIC_TOOLBOX_WITH_ROOT)
  find_package(ROOT REQUIRED COMPONENTS Core RIO Hist Tree TreePlayer Matrix Graf Gpad MathCore)
  addToolboxModule(Root Vector Utils Fs Log)
  linkToolboxExternalLibraries(Root ROOT::Core ROOT::RIO ROOT::Hist ROOT::Tree
    ROOT::TreePlayer ROOT::Matrix ROOT::Graf ROOT::Gpad ROOT::MathCore)
endif()
if(GENERIC_TOOLBOX_WITH_JSON)
  if(NOT TARGET nlohmann_json::nlohmann_json)
    find_package(nlohmann_json REQUIRED CONFIG)
  endif()
  addToolboxModule(Json Vector String Fs Os Utils Log)
  linkToolboxExternalLibraries(Json nlohmann_json::nlohmann_json)
endif()
if(GENERIC_TOOLBOX_WITH_YAML)
  if(NOT TARGET yaml-cpp::yaml-cpp)
    find_package(yaml-cpp REQUIRED CONFIG)
  endif()
  addToolboxModule(Yaml Fs String)
  linkToolboxExternalLibraries(Yaml yaml-cpp::yaml-cpp)
endif()

if(GENERIC_TOOLBOX_BUILD_EXAMPLES)
  add_executable(GenericToolboxExample example/main.cpp)
  target_link_libraries(GenericToolboxExample PRIVATE GenericToolbox::GenericToolbox)
  if(GENERIC_TOOLBOX_WITH_ROOT)
    target_compile_definitions(GenericToolboxExample PRIVATE ENABLE_ROOT_EXTENSION)
  endif()
endif()
if(GENERIC_TOOLBOX_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

if(GENERIC_TOOLBOX_INSTALL_PACKAGE)
  install(DIRECTORY include/ DESTINATION include)
  install(EXPORT GenericToolboxTargets NAMESPACE GenericToolbox:: DESTINATION lib/cmake/GenericToolbox)
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/GenericToolboxConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/GenericToolboxConfig.cmake
    INSTALL_DESTINATION lib/cmake/GenericToolbox)
  install(FILES ${CMAKE_CURRENT_BINARY_DIR}/GenericToolboxConfig.cmake DESTINATION lib/cmake/GenericToolbox)
endif()
