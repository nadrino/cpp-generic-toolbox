foreach(extension IN ITEMS ROOT JSON YAML)
  if(GENERIC_TOOLBOX_WITH_${extension})
    target_compile_definitions(GenericToolboxModules PRIVATE TOOLBOX_TEST_${extension})
  endif()
endforeach()
add_test(NAME GenericToolboxModules COMMAND GenericToolboxModules)

# Independently exercise direct header inclusion, with no compiled definitions.
add_executable(GenericToolboxHeaderOnly Modules.cpp ModulePeer.cpp)
target_include_directories(GenericToolboxHeaderOnly PRIVATE ../include)
set_target_properties(GenericToolboxHeaderOnly PROPERTIES
  CXX_STANDARD 14 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(GenericToolboxHeaderOnly PRIVATE Threads::Threads)
add_test(NAME GenericToolboxHeaderOnly COMMAND GenericToolboxHeaderOnly)
# The two processes write the same scratch file in separate working directories.
file(MAKE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/modules ${CMAKE_CURRENT_BINARY_DIR}/header-only)
set_tests_properties(GenericToolboxModules PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/modules)
set_tests_properties(GenericToolboxHeaderOnly PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/header-only)

# Compile every enabled public module header without relying on include order.
set(headers Macro Log String Os Fs Vector Wrappers Utils Time Thread Stream Map Loops ZipIterator Misc)
foreach(extension IN ITEMS Root Json Yaml)
  string(TOUPPER ${extension} extensionUpper)
  if(GENERIC_TOOLBOX_WITH_${extensionUpper})
    list(APPEND headers ${extension})
  endif()
endforeach()
foreach(header IN LISTS headers)
  set(source ${CMAKE_CURRENT_BINARY_DIR}/Standalone${header}.cpp)
  file(GENERATE OUTPUT ${source} CONTENT "#include \"GenericToolbox.${header}.h\"\n")
  list(APPEND standaloneSources ${source})
endforeach()
add_library(GenericToolboxStandaloneHeaders OBJECT ${standaloneSources})
target_link_libraries(GenericToolboxStandaloneHeaders PRIVATE GenericToolbox::GenericToolbox)
